cmake_minimum_required(VERSION 3.25)
project(CortexStudio VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(cortex_rtp src/transport/H264RtpReceiver.cpp)
target_include_directories(cortex_rtp PUBLIC include)
add_library(cortex_discovery_protocol src/discovery/HandshakeProtocol.cpp)
target_include_directories(cortex_discovery_protocol PUBLIC include)
if(MSVC)
    add_compile_options(/W4 /permissive- /EHsc)
endif()
if(BUILD_TESTING)
    add_executable(rtp_tests tests/RtpTests.cpp)
    target_link_libraries(rtp_tests PRIVATE cortex_rtp)
    add_test(NAME rtp_tests COMMAND rtp_tests)
    add_executable(handshake_tests tests/HandshakeTests.cpp)
    target_link_libraries(handshake_tests PRIVATE cortex_discovery_protocol)
    add_test(NAME handshake_tests COMMAND handshake_tests)
endif()
option(CORTEX_BUILD_RECEIVER "Build Windows D3D11 hardware receiver" ON)
if(CORTEX_BUILD_RECEIVER)
    if(NOT WIN32)
        message(FATAL_ERROR "The receiver requires Windows; use -DCORTEX_BUILD_RECEIVER=OFF for portable RTP tests")
    endif()
    set(FFMPEG_ROOT "" CACHE PATH "FFmpeg development distribution containing include, lib and bin")
    find_path(FFMPEG_INCLUDE libavcodec/avcodec.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
    find_library(AVCODEC_LIBRARY avcodec HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    find_library(AVUTIL_LIBRARY avutil HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    find_library(AVFORMAT_LIBRARY avformat HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    add_library(cortex_receiver
        src/gpu/GpuDevice.cpp src/decoding/HardwareH264Decoder.cpp
        src/transport/UdpH264Source.cpp src/sources/VideoSourceManager.cpp
        src/discovery/StudioDiscovery.cpp)
    target_include_directories(cortex_receiver PUBLIC include)
    target_include_directories(cortex_receiver SYSTEM PRIVATE "${FFMPEG_INCLUDE}")
    target_compile_definitions(cortex_receiver PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(cortex_receiver PUBLIC cortex_rtp cortex_discovery_protocol d3d11 dxgi ws2_32 dnsapi PRIVATE "${AVCODEC_LIBRARY}" "${AVUTIL_LIBRARY}")
    add_executable(cortex-receive apps/receiver_main.cpp)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/pipeline.hlsl" CORTEX_HLSL)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/pipeline.hlsl)
    configure_file(shaders/ShaderSource.h.in generated/ShaderSource.generated.h @ONLY)
    add_library(cortex_processing
        src/filters/GpuColorConverter.cpp src/filters/GpuImageTransform.cpp src/filters/VideoProcessor.cpp
        src/filters/RvmMatting.cpp src/filters/BackgroundVideo.cpp src/output/UnityCameraOutput.cpp)
    target_include_directories(cortex_processing PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" third_party/unitycapture)
    target_include_directories(cortex_processing SYSTEM PRIVATE "${FFMPEG_INCLUDE}")
    target_link_libraries(cortex_processing PUBLIC cortex_receiver PRIVATE d3dcompiler windowscodecs ole32 "${AVFORMAT_LIBRARY}" "${AVCODEC_LIBRARY}" "${AVUTIL_LIBRARY}")
    set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted Microsoft.ML.OnnxRuntime.DirectML NuGet package")
    set(DIRECTML_ROOT "" CACHE PATH "Extracted Microsoft.AI.DirectML NuGet package")
    if(ONNXRUNTIME_ROOT)
        if(NOT EXISTS "${DIRECTML_ROOT}/include/DirectML.h")
            message(FATAL_ERROR "DIRECTML_ROOT must point to the Microsoft.AI.DirectML package")
        endif()
        target_compile_definitions(cortex_processing PRIVATE CORTEX_HAS_DIRECTML)
        target_include_directories(cortex_processing SYSTEM PRIVATE "${ONNXRUNTIME_ROOT}/build/native/include" "${DIRECTML_ROOT}/include")
        target_link_libraries(cortex_processing PRIVATE "${ONNXRUNTIME_ROOT}/runtimes/win-x64/native/onnxruntime.lib")
        file(GLOB ORT_DLLS "${ONNXRUNTIME_ROOT}/runtimes/win-x64/native/*.dll")
        list(APPEND ORT_DLLS "${DIRECTML_ROOT}/bin/x64-win/DirectML.dll")
    endif()
    target_link_libraries(cortex-receive PRIVATE cortex_processing)
    add_library(CortexStudioNative SHARED src/ui/StudioBridge.cpp)
    target_compile_definitions(CortexStudioNative PRIVATE CORTEX_STUDIO_NATIVE_BUILD)
    target_link_libraries(CortexStudioNative PRIVATE cortex_processing d3dcompiler)
    set_target_properties(CortexStudioNative PROPERTIES OUTPUT_NAME CortexStudioNative)
    if(ORT_DLLS)
        add_custom_command(TARGET cortex-receive POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_DLLS} "$<TARGET_FILE_DIR:cortex-receive>" COMMAND_EXPAND_LISTS)
    endif()
    if(EXISTS "${FFMPEG_ROOT}/bin")
        file(GLOB FFMPEG_DLLS "${FFMPEG_ROOT}/bin/*.dll")
        if(FFMPEG_DLLS)
            add_custom_command(TARGET cortex-receive POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:cortex-receive>" COMMAND_EXPAND_LISTS)
        endif()
    endif()
    if(BUILD_TESTING)
        add_executable(studio_autoconnect_tests tests/StudioAutoConnectTests.cpp)
        target_include_directories(studio_autoconnect_tests PRIVATE include)
        target_link_libraries(studio_autoconnect_tests PRIVATE CortexStudioNative ws2_32)
        add_test(NAME studio_autoconnect_tests COMMAND studio_autoconnect_tests)
        set_tests_properties(studio_autoconnect_tests PROPERTIES TIMEOUT 65 RUN_SERIAL TRUE)
        add_executable(manager_tests tests/ManagerTests.cpp)
        target_link_libraries(manager_tests PRIVATE cortex_receiver)
        add_test(NAME manager_tests COMMAND manager_tests)
        add_executable(processing_tests tests/ProcessingTests.cpp)
        target_include_directories(processing_tests PRIVATE third_party/unitycapture)
        target_link_libraries(processing_tests PRIVATE cortex_processing)
        if(ORT_DLLS)
            add_custom_command(TARGET processing_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_DLLS} "$<TARGET_FILE_DIR:processing_tests>" COMMAND_EXPAND_LISTS)
        endif()
        if(FFMPEG_DLLS)
            add_custom_command(TARGET processing_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:processing_tests>" COMMAND_EXPAND_LISTS)
        endif()
        add_test(NAME processing_tests COMMAND processing_tests)
        add_executable(hardware_decoder_tests tests/HardwareDecoderTests.cpp)
        target_link_libraries(hardware_decoder_tests PRIVATE cortex_receiver)
        target_compile_definitions(hardware_decoder_tests PRIVATE CORTEX_TEST_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        add_test(NAME hardware_decoder_tests COMMAND hardware_decoder_tests)
    endif()
endif()
